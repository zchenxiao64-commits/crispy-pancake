function makeDraggable(el, handle){
  let sx,sy,ox,oy,drag=false;
  handle.addEventListener('mousedown',e=>{
    drag=true; el.classList.add('dragging');
    sx=e.clientX; sy=e.clientY;
    const r=el.getBoundingClientRect();
    ox=r.left; oy=r.top;
    e.preventDefault();
  });
  window.addEventListener('mousemove',e=>{
    if(!drag) return;
    let nx=ox+(e.clientX-sx), ny=oy+(e.clientY-sy);
    nx=Math.max(8,Math.min(window.innerWidth-el.offsetWidth-8,nx));
    ny=Math.max(8,Math.min(window.innerHeight-el.offsetHeight-8,ny));
    el.style.left=nx+'px'; el.style.top=ny+'px';
    el.style.right='auto'; el.style.bottom='auto';
  });
  window.addEventListener('mouseup',()=>{ drag=false; el.classList.remove('dragging'); });
  // 触摸
  handle.addEventListener('touchstart',e=>{
    drag=true; el.classList.add('dragging');
    const t=e.touches[0]; sx=t.clientX; sy=t.clientY;
    const r=el.getBoundingClientRect(); ox=r.left; oy=r.top;
  },{passive:true});
  window.addEventListener('touchmove',e=>{
    if(!drag) return; const t=e.touches[0];
    let nx=ox+(t.clientX-sx), ny=oy+(t.clientY-sy);
    nx=Math.max(8,Math.min(window.innerWidth-el.offsetWidth-8,nx));
    ny=Math.max(8,Math.min(window.innerHeight-el.offsetHeight-8,ny));
    el.style.left=nx+'px'; el.style.top=ny+'px'; el.style.right='auto'; el.style.bottom='auto';
  },{passive:true});
  window.addEventListener('touchend',()=>{ drag=false; el.classList.remove('dragging'); });
}

function setupDragScroll(el){
  if(!el) return;
  let down=false,startX,scrollLeft;
  el.addEventListener('mousedown',e=>{ down=true; el.classList.add('grabbing'); startX=e.pageX-el.offsetLeft; scrollLeft=el.scrollLeft; });
  el.addEventListener('mouseleave',()=>{ down=false; el.classList.remove('grabbing'); });
  el.addEventListener('mouseup',()=>{ down=false; el.classList.remove('grabbing'); });
  el.addEventListener('mousemove',e=>{
    if(!down) return; e.preventDefault();
    const x=e.pageX-el.offsetLeft; const walk=(x-startX)*1.4;
    el.scrollLeft=scrollLeft-walk;
  });
}

/* ===================== 滚动渐入 ===================== */
let splashObs=null, fadeObs=null;
function observeSplash(){
  if(splashObs) splashObs.disconnect();
  splashObs=new IntersectionObserver(es=>{
    es.forEach(e=>{ if(e.isIntersecting) e.target.classList.add('in'); });
  },{threshold:0.25});
  document.querySelectorAll('.splash').forEach(el=>{ if(el.offsetParent!==null) splashObs.observe(el); });
}
function observeFade(){
  if(fadeObs) fadeObs.disconnect();
  fadeObs=new IntersectionObserver(es=>{
    es.forEach(e=>{ if(e.isIntersecting) e.target.classList.add('in'); });
  },{threshold:0.1});
  document.querySelectorAll('.fade-sec').forEach(el=>fadeObs.observe(el));
}

/* ===================== 首页 slogan 打字机 ===================== */
function typewriter(el, startDelay=0){
  const text = el.textContent;
  el.innerHTML = '';
  const chars = text.split('');
  chars.forEach((ch)=>{
    const span = document.createElement('span');
    span.className = 'type-char';
    span.textContent = ch === ' ' ? '\u00A0' : ch;
    el.appendChild(span);
  });
  const spans = el.querySelectorAll('.type-char');
  const step = 32;
  spans.forEach((sp, i)=>{
    setTimeout(()=> sp.classList.add('visible'), startDelay + i * step);
  });
  return startDelay + chars.length * step + 260;
}
function initTypewriter(){
  const intro = document.querySelector('.home-right .intro');
  const quote = document.querySelector('.home-right .quote');
  if(!intro || intro.dataset.typed === '1') return;
  intro.dataset.typed = '1';
  const finishAt = typewriter(intro, 420);
  if(quote){
    setTimeout(()=> quote.classList.add('show'), finishAt + 120);
  }
}
